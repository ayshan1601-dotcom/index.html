    <!-- Stage 1: The Surprise Reveal Box -->
    <div class="envelope-container" id="surpriseBox" onclick="revealMessage()">
        <div class="envelope">✉️💝</div>
        <div class="click-text">You have a special message. Click to open!</div>
    </div>

    <!-- Stage 2: The Apology Content Card -->
    <div class="card" id="mainCard">
        <h1 id="title">I'm So Sorry, Handsome... 🥺</h1>
        <p id="message">I really hate it when we fight, and I hate hurting you even more. You mean the absolute world to me. Can you please forgive your favorite headache?</p>
        
        <div class="btn-container">
            <button class="btn" id="yesBtn" onclick="forgiven()">Yes</button>
            <button class="btn" id="noBtn" onmouseover="runAway()">No</button>
        </div>
    </div>

    <script>
        // 1. Reveal Stage: Transition from Envelope to Card
        function revealMessage() {
            const box = document.getElementById('surpriseBox');
            const card = document.getElementById('mainCard');
            
            box.style.display = 'none';
            card.style.display = 'block';
            
            // Trigger floating heart explosion effect
            for(let i=0; i<30; i++) { createHeart(); }

            setTimeout(() => {
                card.style.opacity = '1';
                card.style.transform = 'scale(1)';
            }, 50);
        }

        // 2. Interactive "No" Button: Runs away on hover
        function runAway() {
            const noBtn = document.getElementById('noBtn');
            // Calculate a completely random position inside the safe browser window boundaries
            const x = Math.random() * (window.innerWidth - noBtn.offsetWidth - 40) + 20;
            const y = Math.random() * (window.innerHeight - noBtn.offsetHeight - 40) + 20;
            
            noBtn.style.position = 'fixed';
            noBtn.style.left = x + 'px';
            noBtn.style.top = y + 'px';
        }

        // 3. Success Stage: If he managed or wanted to click Yes
        function forgiven() {
            document.getElementById('title').innerHTML = "YAY! Thank you! ❤️";
            document.getElementById('message').innerHTML = "I promise to make it up to you. I love you to the moon and back! See you soon!";
            document.getElementById('yesBtn').style.display = 'none';
            document.getElementById('noBtn').style.display = 'none';
            
            // Continuous happy heart rain
            setInterval(createHeart, 150);
        }

        // Helper function for visual flair
        function createHeart() {
            const heart = document.createElement('div');
            heart.classList.add('heart-particle');
            heart.innerHTML = ['❤️','💖','💝','🌸'][Math.floor(Math.random()*4)];
            heart.style.left = Math.random() * 100 + 'vw';
            heart.style.top = Math.random() * 100 + 'vh';
            heart.style.fontSize = Math.random() * 20 + 20 + 'px';
            heart.style.transition = 'all 1.5s ease-out';
            
            document.body.appendChild(heart);
            
            setTimeout(() => {
                heart.style.transform = 'translateY(-150px) scale(0)';
                heart.style.opacity = '0';
            }, 50);
            
            setTimeout(() => { heart.remove(); }, 1500);
        }
    </script>
